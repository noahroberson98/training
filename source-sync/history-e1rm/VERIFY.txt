# Run from the extracted review root. Do not publish.
# $PWD is captured before any cd. The patch redirect uses that root, not the directory entered by cd.

REVIEW_ROOT="$(pwd)"
test -f "$REVIEW_ROOT/patches/persist-save-status.patch"
test -f "$REVIEW_ROOT/scripts/template.5ad77b37.html"
test -f "$REVIEW_ROOT/scripts/preview_current.py"
rm -rf /tmp/gen-review
mkdir -p /tmp/gen-review/scripts
cp "$REVIEW_ROOT/scripts/template.5ad77b37.html" /tmp/gen-review/scripts/template.html
(cd /tmp/gen-review && patch -p1 --fuzz=0 < "$REVIEW_ROOT/patches/persist-save-status.patch")
echo PATCH_EXIT:$?
sha256sum /tmp/gen-review/scripts/template.html
python3 "$REVIEW_ROOT/scripts/reconstruct_review.py" /tmp/gen-review/scripts/template.html "$REVIEW_ROOT/before/index.html" /tmp/rebuilt-review.html
echo REBUILD_EXIT:$?
cmp /tmp/rebuilt-review.html "$REVIEW_ROOT/corrected/index.html"
echo CMP_EXIT:$?
python3 "$REVIEW_ROOT/tests/test_e1rm_persist.py" "$REVIEW_ROOT/before/index.html" /tmp/recipe-persist-before
echo PERSIST_BEFORE_EXIT:$?
python3 "$REVIEW_ROOT/tests/test_e1rm_persist.py" "$REVIEW_ROOT/corrected/index.html" /tmp/recipe-persist-after
echo PERSIST_AFTER_EXIT:$?
python3 "$REVIEW_ROOT/tests/test_e1rm_reach.py" "$REVIEW_ROOT/corrected/index.html" /tmp/recipe-reach
echo REACH_EXIT:$?
python3 "$REVIEW_ROOT/tests/test_e1rm_exact.py" "$REVIEW_ROOT/corrected/index.html" /tmp/recipe-exact
echo EXACT_EXIT:$?
HISTORY_PATCH="$REVIEW_ROOT/patches/template.html.patch" python3 "$REVIEW_ROOT/tests/test_e1rm_history.py"
echo HISTORY_EXIT:$?
